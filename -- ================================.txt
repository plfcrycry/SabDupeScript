-- Check Key
CheckBtn.MouseButton1Click:Connect(function()
    local entered = KeyInput.Text:upper():gsub("%s+", "")
    if entered == "" then
        setStatus("⛔  Please enter a key first.", Color3.fromRGB(248, 113, 113))
        return
    end
    local valid = false
    for _, k in ipairs(VALID_KEYS) do
        if entered == k:upper() then valid = true break end
    end
    if valid then
        setStatus("✅  Key valid! Loading script...", Color3.fromRGB(74, 222, 128))
        -- ✏️ PUT YOUR SCRIPT EXECUTION HERE:
        -- loadstring(game:HttpGet("YOUR_SCRIPT_URL"))()
    else
        setStatus("❌  Invalid key. Get one via the link!", Color3.fromRGB(248, 113, 113))
    end
end)
